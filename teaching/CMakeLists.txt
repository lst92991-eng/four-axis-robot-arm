cmake_minimum_required(VERSION 3.18)
project(scara_teaching LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "SocketCAN/GPIO入口在Linux/WSL构建；Windows可用VS Code阅读源码")
endif()
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs objdetect geometry)
find_package(Threads REQUIRED)
add_library(scara_core STATIC
    robot/can.cpp robot/motor.cpp robot/magnet.cpp
    motion/kinematics.cpp motion/motion.cpp motion/filter.cpp task/pick_place.cpp
    vision/camera.cpp vision/board_vision.cpp
    chess/board.cpp chess/pikafish.cpp app/config.cpp app/journal.cpp)
target_include_directories(scara_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(scara_core PUBLIC ${OpenCV_LIBS} Threads::Threads)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(scara_core PUBLIC stdc++fs)
endif()
target_compile_options(scara_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(scara_robot main.cpp)
target_link_libraries(scara_robot PRIVATE scara_core)
target_compile_options(scara_robot PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(SCARA_WITH_RKNN "Link actual RV1106 RKNN runtime" OFF)
if(SCARA_WITH_RKNN)
    find_path(RKNN_INCLUDE_DIR rknn_api.h PATHS ${RKNN_SDK_DIR}/include NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    find_library(RKNN_LIBRARY rknnmrt PATHS ${RKNN_SDK_DIR}/lib NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    target_sources(scara_robot PRIVATE vision/detector.cpp)
    target_include_directories(scara_robot PRIVATE ${RKNN_INCLUDE_DIR})
    target_link_libraries(scara_robot PRIVATE ${RKNN_LIBRARY})
    target_compile_definitions(scara_robot PRIVATE SCARA_WITH_RKNN=1)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(scara_tests tests/main.cpp tests/device_tests.cpp tests/motion_tests.cpp tests/vision_chess_tests.cpp tests/app_tests.cpp)
    target_link_libraries(scara_tests PRIVATE scara_core)
    target_compile_definitions(scara_tests PRIVATE SCARA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_options(scara_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME teaching_core COMMAND scara_tests)
endif()
