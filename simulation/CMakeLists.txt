cmake_minimum_required(VERSION 3.18)
project(scara_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs objdetect geometry highgui)
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/deps-raylib-5.5")
find_package(raylib 5.5.0 EXACT CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
option(SIM_DEBUG "Build removable debug callbacks" ON)
add_executable(scara_sim sim.cpp cad_view.cpp yolo_gpu.cpp x42s.cpp motion_filter.cpp
                          chess/board.cpp chess/pikafish.cpp)
target_include_directories(scara_sim PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(scara_sim PRIVATE ${OpenCV_LIBS} raylib OpenGL::GL)
target_compile_options(scara_sim PRIVATE -Wall -Wextra -Wpedantic)
if(SIM_DEBUG)
    target_sources(scara_sim PRIVATE debug.cpp)
    target_compile_definitions(scara_sim PRIVATE SIM_DEBUG=1)
endif()
option(SIM_DRIVER_CHECKS "Build the isolated vcan acceptance runner" OFF)
if(SIM_DRIVER_CHECKS)
    add_executable(driver_check tests/driver_check.cpp x42s.cpp motion_filter.cpp)
    target_include_directories(driver_check PRIVATE . ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(driver_check PRIVATE ${OpenCV_LIBS})
    target_compile_options(driver_check PRIVATE -Wall -Wextra -Wpedantic)
endif()
