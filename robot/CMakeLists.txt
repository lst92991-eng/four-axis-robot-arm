cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(scara_motor_stage LANGUAGES CXX)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "CAN模块需要Linux SocketCAN；Windows面板仍使用原入口")
endif()
find_package(Threads REQUIRED)
add_library(scara_motor STATIC can.cpp motor.cpp)
target_include_directories(scara_motor PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_compile_features(scara_motor PUBLIC cxx_std_17)
target_compile_options(scara_motor PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(scara_motor PUBLIC Threads::Threads)
add_executable(scara_motor_test ../app/motor_test.cpp)
target_link_libraries(scara_motor_test PRIVATE scara_motor)
target_compile_options(scara_motor_test PRIVATE -Wall -Wextra -Wpedantic)
