cmake_minimum_required(VERSION 3.20)
project(scara_vision_geometry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs geometry calib)
add_library(scara_geometry vision/geometry.cpp vision/board.cpp)
target_include_directories(scara_geometry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(scara_geometry PUBLIC ${OpenCV_LIBS})
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(scara_geometry PUBLIC stdc++fs)
endif()
if(MSVC)
  target_compile_options(scara_geometry PUBLIC /utf-8)
else()
  target_compile_options(scara_geometry PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(scara_geometry_tool app/geometry_tool.cpp)
target_link_libraries(scara_geometry_tool PRIVATE scara_geometry)
add_library(scara_chess chess/board.cpp)
target_include_directories(scara_chess PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(scara_chess PUBLIC /utf-8)
else()
  target_compile_options(scara_chess PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(scara_game_geometry chess/vision_bridge.cpp)
target_link_libraries(scara_game_geometry PUBLIC scara_chess scara_geometry)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # POSIX subprocess transport; no motor or CAN calls. Keep CAN supervision independent.
  add_library(scara_pikafish chess/pikafish.cpp)
  target_include_directories(scara_pikafish PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(scara_pikafish PUBLIC scara_chess)
  target_compile_options(scara_pikafish PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
  target_link_options(scara_geometry_tool PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(scara_geometry_test tests/geometry_test.cpp)
  target_link_libraries(scara_geometry_test PRIVATE scara_geometry)
  add_test(NAME geometry_contracts COMMAND scara_geometry_test)
  add_executable(scara_geometry_image_test tests/image_geometry_test.cpp)
  target_link_libraries(scara_geometry_image_test PRIVATE scara_geometry)
  add_executable(scara_game_geometry_test tests/game_geometry_test.cpp)
  target_link_libraries(scara_game_geometry_test PRIVATE scara_game_geometry)
  add_test(NAME game_geometry_contracts COMMAND scara_game_geometry_test)
endif()
if(WIN32 AND TARGET opencv_world)
  foreach(app scara_geometry_tool scara_geometry_test scara_geometry_image_test scara_game_geometry_test)
    if(TARGET ${app})
      add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:opencv_world> $<TARGET_FILE_DIR:${app}>)
    endif()
  endforeach()
endif()
# This project intentionally links no firmware, CAN transport, or motor library.
