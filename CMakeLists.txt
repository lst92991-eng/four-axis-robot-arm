cmake_minimum_required(VERSION 3.18)
project(scara_robot LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs objdetect geometry)
find_package(Threads REQUIRED)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_subdirectory(robot)
endif()

add_subdirectory(algorithm)
add_library(scara_game STATIC chess/board.cpp chess/pikafish.cpp)
target_include_directories(scara_game PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(scara_simulator STATIC robot/simulator.cpp)
target_include_directories(scara_simulator PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(scara_simulator PUBLIC scara_algorithm scara_game)

add_executable(scara_robot main.cpp vision/board_vision.cpp vision/camera.cpp)
target_include_directories(scara_robot PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(scara_robot PRIVATE scara_game scara_simulator ${OpenCV_LIBS} Threads::Threads)
target_compile_options(scara_robot PRIVATE -Wall -Wextra -Wpedantic)

# 同一主程序：电脑可运行虚拟机械臂；板端构建时显式启用真实RKNN API。
option(SCARA_WITH_RKNN "Build the real RV1106 NPU detector" OFF)
if(SCARA_WITH_RKNN)
    find_path(RKNN_INCLUDE_DIR rknn_api.h PATHS ${RKNN_SDK_DIR}/include NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    find_library(RKNN_LIBRARY NAMES rknnmrt PATHS ${RKNN_SDK_DIR}/lib NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    target_sources(scara_robot PRIVATE vision/detector.cpp)
    target_include_directories(scara_robot PRIVATE ${RKNN_INCLUDE_DIR})
    target_link_libraries(scara_robot PRIVATE ${RKNN_LIBRARY})
    target_compile_definitions(scara_robot PRIVATE SCARA_WITH_RKNN=1)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(scara_tests tests/main.cpp tests/algorithm_tests.cpp tests/chess_tests.cpp tests/simulator_tests.cpp tests/commands_tests.cpp tests/vision_tests.cpp vision/board_vision.cpp)
    target_include_directories(scara_tests PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(scara_tests PRIVATE scara_simulator ${OpenCV_LIBS})
    target_compile_options(scara_tests PRIVATE -Wall -Wextra -Wpedantic)
    if(TARGET scara_motor)
        target_sources(scara_tests PRIVATE tests/motor_tests.cpp)
        target_link_libraries(scara_tests PRIVATE scara_motor)
        target_compile_definitions(scara_tests PRIVATE SCARA_HAS_MOTOR=1)
    endif()
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
        target_link_libraries(scara_tests PRIVATE stdc++fs)
    endif()
    add_test(NAME scara_core COMMAND scara_tests)
endif()
