cmake_minimum_required(VERSION 3.20)
project(scara_vision_geometry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs geometry calib)
add_library(scara_geometry vision/geometry.cpp vision/board.cpp)
target_include_directories(scara_geometry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(scara_geometry PUBLIC ${OpenCV_LIBS})
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(scara_geometry PUBLIC stdc++fs)
endif()
if(MSVC)
  target_compile_options(scara_geometry PUBLIC /utf-8)
else()
  target_compile_options(scara_geometry PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(scara_geometry_tool app/geometry_tool.cpp)
target_link_libraries(scara_geometry_tool PRIVATE scara_geometry)
add_library(scara_chess chess/board.cpp)
target_include_directories(scara_chess PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(scara_chess PUBLIC /utf-8)
else()
  target_compile_options(scara_chess PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(scara_game_geometry chess/vision_bridge.cpp)
target_link_libraries(scara_game_geometry PUBLIC scara_chess scara_geometry)

option(SCARA_BUILD_VISION "Build the camera and YOLO processing stage" ON)
option(SCARA_WITH_RKNN "Build the RV1106 native NPU application" OFF)
if(SCARA_BUILD_VISION)
  find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs videoio)
  find_package(Threads REQUIRED)
  add_library(scara_vision_core vision/yolo.cpp vision/capture.cpp)
  target_include_directories(scara_vision_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(scara_vision_core PUBLIC ${OpenCV_LIBS} Threads::Threads)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(scara_vision_core PUBLIC stdc++fs)
  endif()
  if(MSVC)
    target_compile_options(scara_vision_core PUBLIC /utf-8)
  else()
    target_compile_options(scara_vision_core PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(SCARA_WITH_RKNN)
    find_path(RKNN_INCLUDE_DIR rknn_api.h HINTS ${RKNN_SDK_DIR}/include NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    find_library(RKNN_LIBRARY NAMES rknnmrt HINTS ${RKNN_SDK_DIR}/lib NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    add_executable(scara_vision app/vision_main.cpp vision/yolo_npu.cpp)
    target_include_directories(scara_vision PRIVATE ${RKNN_INCLUDE_DIR})
    target_link_libraries(scara_vision PRIVATE scara_vision_core ${RKNN_LIBRARY})
    install(TARGETS scara_vision RUNTIME DESTINATION bin)
    install(PROGRAMS tools/run-vision.sh DESTINATION bin)
    install(FILES config/vision/chinese-labels.png DESTINATION config/vision)
  endif()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # POSIX subprocess transport; no motor or CAN calls. Keep CAN supervision independent.
  add_library(scara_pikafish chess/pikafish.cpp)
  target_include_directories(scara_pikafish PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(scara_pikafish PUBLIC scara_chess)
  target_compile_options(scara_pikafish PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
  target_link_options(scara_geometry_tool PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(scara_geometry_test tests/geometry_test.cpp)
  target_link_libraries(scara_geometry_test PRIVATE scara_geometry)
  add_test(NAME geometry_contracts COMMAND scara_geometry_test)
  add_executable(scara_geometry_image_test tests/image_geometry_test.cpp)
  target_link_libraries(scara_geometry_image_test PRIVATE scara_geometry)
  add_executable(scara_game_geometry_test tests/game_geometry_test.cpp)
  target_link_libraries(scara_game_geometry_test PRIVATE scara_game_geometry)
  add_test(NAME game_geometry_contracts COMMAND scara_game_geometry_test)
  if(SCARA_BUILD_VISION)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_executable(scara_vision_test tests/yolo_pipeline_test.cpp)
    target_link_libraries(scara_vision_test PRIVATE scara_vision_core)
    add_executable(scara_vision_replay tests/vision_replay.cpp)
    target_link_libraries(scara_vision_replay PRIVATE scara_vision_core)
    add_test(NAME vision_pipeline_contracts COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/with_camera_stream.py $<TARGET_FILE:scara_vision_test>)
    set_tests_properties(vision_pipeline_contracts PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
    if(RKNN_INCLUDE_DIR)
      add_executable(scara_rknn_test tests/rknn_runtime_test.cpp vision/yolo_npu.cpp)
      target_include_directories(scara_rknn_test PRIVATE ${RKNN_INCLUDE_DIR})
      target_link_libraries(scara_rknn_test PRIVATE scara_vision_core)
      add_test(NAME rknn_adapter_contracts COMMAND scara_rknn_test)
      add_executable(scara_vision_entry_test app/vision_main.cpp vision/yolo_npu.cpp tests/rknn_runtime_test.cpp)
      target_compile_definitions(scara_vision_entry_test PRIVATE RKNN_MOCK_ENTRY=1)
      target_include_directories(scara_vision_entry_test PRIVATE ${RKNN_INCLUDE_DIR})
      target_link_libraries(scara_vision_entry_test PRIVATE scara_vision_core)
      add_test(NAME vision_entry_contracts COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/with_camera_stream.py $<TARGET_FILE:scara_vision_entry_test> --entry)
      set_tests_properties(vision_entry_contracts PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
        DEPENDS vision_pipeline_contracts)
    endif()
  endif()
endif()
if(WIN32 AND TARGET opencv_world)
  foreach(app scara_geometry_tool scara_geometry_test scara_geometry_image_test scara_game_geometry_test scara_vision_test scara_vision_replay scara_rknn_test scara_vision_entry_test)
    if(TARGET ${app})
      add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:opencv_world> $<TARGET_FILE_DIR:${app}>)
    endif()
  endforeach()
  get_target_property(opencv_dll opencv_world IMPORTED_LOCATION_RELEASE)
  get_filename_component(opencv_bin ${opencv_dll} DIRECTORY)
  foreach(app scara_vision_test scara_vision_entry_test)
    if(TARGET ${app} AND EXISTS "${opencv_bin}/opencv_videoio_ffmpeg500_64.dll")
      add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${opencv_bin}/opencv_videoio_ffmpeg500_64.dll" $<TARGET_FILE_DIR:${app}>)
    endif()
  endforeach()
endif()
